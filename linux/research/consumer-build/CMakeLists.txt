cmake_minimum_required(VERSION 3.22)
project(MadSenseConsumerProbes LANGUAGES C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "These observers require Linux device APIs")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2>=2.24)
add_executable(consumer_sdl2 ../consumer_sdl.c)
target_compile_definitions(consumer_sdl2 PRIVATE CONSUMER_SDL_MAJOR=2)
target_link_libraries(consumer_sdl2 PRIVATE PkgConfig::SDL2)

if(NOT EXISTS "${SDL_SOURCE_DIRECTORY}/CMakeLists.txt")
  message(FATAL_ERROR "SDL_SOURCE_DIRECTORY must identify the verified pinned SDL3 source")
endif()
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SDL_AUDIO OFF CACHE BOOL "" FORCE)
set(SDL_VIDEO OFF CACHE BOOL "" FORCE)
set(SDL_RENDER OFF CACHE BOOL "" FORCE)
set(SDL_GPU OFF CACHE BOOL "" FORCE)
set(SDL_CAMERA OFF CACHE BOOL "" FORCE)
set(SDL_HIDAPI_LIBUSB OFF CACHE BOOL "" FORCE)
add_subdirectory("${SDL_SOURCE_DIRECTORY}" sdl3 EXCLUDE_FROM_ALL)
add_executable(consumer_sdl3 ../consumer_sdl.c)
target_compile_definitions(consumer_sdl3 PRIVATE CONSUMER_SDL_MAJOR=3)
target_link_libraries(consumer_sdl3 PRIVATE SDL3::SDL3-static)
add_executable(ff_probe ../ff_probe.c)
foreach(target consumer_sdl2 consumer_sdl3 ff_probe)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
endforeach()
