cmake_minimum_required(VERSION 3.20)
set(CMAKE_OSX_DEPLOYMENT_TARGET "15.0" CACHE STRING "Minimum macOS version")
project(MadSenseAudio LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
find_package(Threads REQUIRED)
add_executable(madsense-audio-ring-test ring_test.cpp)
target_link_libraries(madsense-audio-ring-test PRIVATE Threads::Threads)
add_test(NAME audio-ring COMMAND madsense-audio-ring-test)
add_executable(madsense-audio-resync-test resync_test.cpp)
target_link_libraries(madsense-audio-resync-test PRIVATE Threads::Threads)
add_test(NAME audio-resync COMMAND madsense-audio-resync-test)

if(APPLE)
  set(AUDIO_FRAMEWORKS "-framework CoreAudio" "-framework CoreFoundation")
  set(DRIVER_BUNDLE "${CMAKE_CURRENT_BINARY_DIR}/MadSenseAudio.driver")
  file(MAKE_DIRECTORY "${DRIVER_BUNDLE}/Contents/MacOS")
  configure_file(Info.plist "${DRIVER_BUNDLE}/Contents/Info.plist" COPYONLY)
  add_library(MadSenseAudio MODULE plugin.cpp)
  set_target_properties(MadSenseAudio PROPERTIES PREFIX "" SUFFIX "" LIBRARY_OUTPUT_DIRECTORY "${DRIVER_BUNDLE}/Contents/MacOS")
  target_link_libraries(MadSenseAudio PRIVATE ${AUDIO_FRAMEWORKS})
  add_library(madsense_audio SHARED consumer.cpp)
  target_link_libraries(madsense_audio PRIVATE ${AUDIO_FRAMEWORKS})
  set_target_properties(madsense_audio PROPERTIES INSTALL_NAME_DIR "@rpath")
  add_executable(madsense-audio-plugin-test plugin_test.cpp plugin.cpp)
  target_link_libraries(madsense-audio-plugin-test PRIVATE ${AUDIO_FRAMEWORKS})
  add_test(NAME audio-plugin-contract COMMAND madsense-audio-plugin-test)
  add_executable(madsense-audio-installed-probe installed_probe.cpp)
  target_link_libraries(madsense-audio-installed-probe PRIVATE madsense_audio ${AUDIO_FRAMEWORKS})
endif()
